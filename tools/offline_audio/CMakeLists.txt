cmake_minimum_required(VERSION 3.24)
project(nte_audio LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(MUSCRIPTOR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_BUILD_BENCH OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_METAL OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_VULKAN OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_CPU_VARIANTS OFF CACHE BOOL "" FORCE)
set(MUSCRIPTOR_GGML_FP32_ACCUM OFF CACHE BOOL "" FORCE)
if(NOT MUSCRIPTOR_SOURCE)
  message(FATAL_ERROR "Set MUSCRIPTOR_SOURCE to the pinned muscriptor.cpp source")
endif()
add_subdirectory(${MUSCRIPTOR_SOURCE}/cpp muscriptor)
# MSVC does not define __F16C__ for /arch:AVX. Our runtime guard already
# requires F16C; without this ggml silently converts every F16 weight scalarly.
if(MSVC AND TARGET ggml-cpu)
  target_compile_definitions(ggml-cpu PRIVATE __F16C__)
endif()
add_executable(nte-audio main.cpp)
target_link_libraries(nte-audio PRIVATE muscriptor_ggml psapi)
if(MSVC)
  target_compile_options(nte-audio PRIVATE /utf-8 /EHsc)
endif()
